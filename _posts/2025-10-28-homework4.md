Sample mean:
$$
f(n) = \frac{X_1 + X_2 + \cdots + X_n}{n}
$$

Law of Large Numbers:
$$
f(n) \to E[X] \quad \text{as } n \to \infty
$$

<script>
function runLLN() {
  const p = parseFloat(document.getElementById('pVal').value);
  const n = parseInt(document.getElementById('nVal').value);
  const m = parseInt(document.getElementById('mVal').value);
  const traces = [];

  for (let i = 0; i < m; i++) {
    const trials = Array.from({length: n}, () => Math.random() < p ? 1 : 0);
    const run = []; let sum = 0;
    for (let k = 0; k < n; k++) { sum += trials[k]; run.push(sum / (k + 1)); }
    traces.push({
      y: run,
      type: 'scatter',
      mode: 'lines',
      name: 'Run ' + (i + 1),
      line: {width: 2}, opacity: 0.8
    });
  }

  traces.push({
    y: Array(n).fill(p),
    type: 'scatter',
    mode: 'lines',
    name: 'True mean (p = ' + p + ')',
    line: {dash: 'dash', color: 'black', width: 2}
  });

  Plotly.newPlot('llnPlot', traces, {
    title: 'Law of Large Numbers Demonstration',
    xaxis: {title: 'Number of trials'},
    yaxis: {title: 'Running mean', range: [0, 1]},
    plot_bgcolor: '#ffffff',
    paper_bgcolor: '#ffffff'
  }, {responsive: true});
}

document.getElementById('runBtn').addEventListener('click', runLLN);
window.addEventListener('load', runLLN);
</script>

<script>
const pHist = 0.3, nHist = 600, runsHist = 1000;
const means = [];

for (let i = 0; i < runsHist; i++) {
  let sum = 0;
  for (let k = 0; k < nHist; k++) sum += (Math.random() < pHist ? 1 : 0);
  means.push(sum / nHist);
}

Plotly.newPlot('histLLN', [{
  x: means,
  type: 'histogram',
  marker: {color: 'cyan', line: {color: 'black', width: 1}},
  nbinsx: 30
}], {
  title: {text: 'Histogram of sample means (p = 0.3, n = 600)'},
  xaxis: {title: 'Sample mean'},
  yaxis: {title: 'Frequency'},
  plot_bgcolor: '#222222',
  paper_bgcolor: '#222222',
  font: {color: 'white'}
}, {responsive: true});
</script>
