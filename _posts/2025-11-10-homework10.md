### 1. Counting Process Poisson

- We consider a time interval $$ T = 1 $$ during which events occur independently and uniformly in time at a constant average rate $$ \lambda $$

- To approximate this process numerically, divide the interval into $$ n $$ small subintervals of equal length
$$ \Delta t = \frac{T}{n}. $$

- In each of these subintervals, generate an event with probability $$ p = \frac{\lambda}{n}. $$

- This creates a discrete-time counting process that approximates a continuous-time process as $$ ( n \to \infty ). $$

---

### 2. What Stochastic Process This Approximates

As $$ ( n ) $$ becomes large, the Bernoulli-counting construction converges to a **Poisson process** with rate $$ ( \lambda ) $$ : $$ \{N(t) : t \in [0,1]\}. $$

For any time \( t \):

$$
N(t) \sim \mathrm{Poisson}(\lambda t).
$$

In particular:

$$
N(1) \sim \mathrm{Poisson}(\lambda).
$$

--- 

### 3. Theoretical Properties of the Limiting Poisson Process

## a. Independent increments

For two disjoint time intervals:

$$
N(t_2) - N(t_1) \quad \text{and} \quad N(t_4) - N(t_3)
$$

are independent.

## b. Stationary increments

Counts depend only on the interval length:

$$
N(t+h) - N(t) \sim \mathrm{Poisson}(\lambda h).
$$

## c. Distribution of inter-arrival times

Let $$ ( T_1, T_2, \dots ) $$ be event times.  
Define inter-arrival times:

$$
X_k = T_k - T_{k-1}.
$$

Then each $$ ( X_k ) $$ is exponentially distributed:

$$
X_k \sim \mathrm{Exponential}(\lambda).
$$

with mean

$$
\mathbb{E}[X_k] = \frac{1}{\lambda}.
$$

---

### 4. Meaning of the Rate Parameter $$ ( \lambda ) $$

The parameter $$ ( \lambda ) $$ represents the expected number of events per unit time:

$$
\mathbb{E}[N(1)] = \lambda.
$$

- Large $$ ( \lambda ) $$ → frequent events, steep sample paths  
- Small $$ ( \lambda ) $$ → rare events, flatter paths

<script>
/* ---------- Utilities ---------- */

// simulate one trajectory
function simulatePath(lambda, T, steps) {
  const dt = T / steps;
  let t = 0, count = 0;
  const times=[0], counts=[0];

  for (let i=0;i<steps;i++){
    if (Math.random() < lambda*dt) count++;
    t += dt;
    times.push(t);
    counts.push(count);
  }
  return {times, counts};
}

// simulate final N(T) values (samples = N_steps)
function simulateSamples(lambda, T, steps){
  const out = new Array(steps);
  const dt = T/steps;

  for (let s=0;s<steps;s++){
    let count = 0;
    for (let i=0;i<steps;i++){
      if (Math.random() < lambda*dt) count++;
    }
    out[s] = count;
  }
  return out;
}

// factorial
function factorial(n){
  if (n<2) return 1;
  let f=1;
  for (let i=2;i<=n;i++) f*=i;
  return f;
}

// Poisson PMF
function poissonPMF(mean,kmax){
  const x = [], y = [];
  for (let k=0;k<=kmax;k++){
    x.push(k);
    y.push(Math.exp(-mean)*Math.pow(mean,k)/factorial(k));
  }
  return {x,y};
}

/* ---------- Plotting ---------- */

function layoutCommon(title){
  return {
    title: { text:title, font:{color:'#e6edf3'} },
    paper_bgcolor:'rgba(0,0,0,0)',
    plot_bgcolor:'rgba(0,0,0,0)',
    font:{ color:'#dbeafe' },
    margin:{t:60,b:60,l:60,r:40},
  };
}

function plotTrajectory(lambda,T,steps){
  const sim = simulatePath(lambda,T,steps);
  const maxC = Math.max(...sim.counts);

  Plotly.react("plot_traj", [{
    x: sim.times,
    y: sim.counts,
    mode:"lines",
    line:{shape:"hv",width:3,color:"#4ea1ff"},
    name:"N(t)"
  }], Object.assign(layoutCommon("Sample Trajectory"), {
    xaxis:{title:"t",range:[0,T]},
    yaxis:{title:"N(t)",range:[0, Math.max(3,Math.ceil(maxC*1.1))]}
  }), {responsive:true});
}

function plotDistribution(lambda,T,steps){
  const arr = simulateSamples(lambda,T,steps);
  const maxObs = Math.max(...arr);
  const mean = lambda*T;

  const kmax = Math.max(maxObs, Math.ceil(mean + 6*Math.sqrt(mean||1)));

  // frequencies
  const freq = new Array(kmax+1).fill(0);
  for (let v of arr) if (v <= kmax) freq[v]++;

  // PMF scaled to histogram counts
  const pmf = poissonPMF(mean, kmax);
  const scaledPMF = pmf.y.map(p => p * steps);

  const ymax = Math.ceil(Math.max(...freq, ...scaledPMF)*1.12);

  const bars = {
    x: pmf.x,
    y: freq,
    type:"bar",
    marker:{color:"#2ea3ff"},
    opacity:0.40,
    name:"Simulated"
  };

  const curve = {
    x: pmf.x,
    y: scaledPMF,
    mode:"lines+markers",
    line:{width:3,color:"#ffd166"},
    marker:{size:6,color:"#ffd166"},
    name:"Poisson(λT)"
  };

  Plotly.react("plot_dist", [bars, curve],
    Object.assign(layoutCommon("Distribution of N(T)"), {
      xaxis:{title:"k (event count)"},
      yaxis:{title:"Frequency",range:[0,ymax]},
      barmode:"overlay"
    }),
    {responsive:true}
  );
}

  const steps = Math.max(100, Number(inp_Nsteps.value));

  plotTrajectory(lambda,T,steps);
  plotDistribution(lambda,T,steps);
}

// ==== Utility: Random Poisson approximation via Bernoulli trials ====
function simulateCounts(lambda, n=5000) {
    let dt = 1/n;
    let t = 0;
    let N = 0;
    let times = [0];
    let counts = [0];
    for (let i = 0; i < n; i++) {
        if (Math.random() < lambda / n) N += 1;
        t += dt;
        times.push(t);
        counts.push(N);
    }
    return {times: times, counts: counts};
}


// ===== 1. TRAJECTORY =====
function genTraj() {
    let lambda = Number(document.getElementById("lambda_traj").value);
    document.getElementById("lambda_traj_val").textContent = "λ = " + lambda;

    let sim = simulateCounts(lambda);

    Plotly.newPlot("traj", [{
        x: sim.times,
        y: sim.counts,
        mode: "lines",
        line: {shape: "hv"}
    }], {
        title: "Sample Trajectory",
        xaxis: {title: "t"},
        yaxis: {title: "N(t)"}
    });
}


// ===== 2. HISTOGRAM OF N(1) =====
function poissonPMF(lambda, kmax=30) {
    let pmf_x = [];
    let pmf_y = [];
    for (let k = 0; k <= kmax; k++) {
        pmf_x.push(k);
        pmf_y.push(Math.exp(-lambda) * Math.pow(lambda, k) / factorial(k));
    }
    return {x: pmf_x, y: pmf_y};
}

function factorial(n) {
    if (n === 0) return 1;
    let f = 1;
    for (let i=1; i<=n; i++) f *= i;
    return f;
}

function genHistN() {
    let lambda = Number(document.getElementById("lambda_hist").value);
    document.getElementById("lambda_hist_val").textContent = "λ = " + lambda;

    let samples = [];
    for (let i=0; i<2000; i++) {
        let s = simulateCounts(lambda);
        samples.push(s.counts[s.counts.length-1]);
    }

    let pmf = poissonPMF(lambda, 30);

    Plotly.newPlot("histN", [
        {
            x: samples,
            type: "histogram",
            opacity: 0.6,
            name: "Simulated"
        },
        {
            x: pmf.x,
            y: pmf.y.map(v => v * 2000),
            mode: "lines+markers",
            name: "Poisson(λ)"
        }
    ], {
        title: "Distribution of N(1)",
        xaxis: {title: "k"},
        yaxis: {title: "Frequency"}
    });
}


// ===== 3. INTERARRIVAL TIMES =====
function simulateInterarrivals(lambda, n=5000) {
    let dt = 1/n;
    let t = 0;
    let arrivals = [];

    for (let i=0; i<n; i++) {
        if (Math.random() < lambda/n) arrivals.push(t);
        t += dt;
    }

    let inter = [];
    for (let i=1; i<arrivals.length; i++) inter.push(arrivals[i] - arrivals[i-1]);

    return inter;
}

function genInter() {
    let lambda = Number(document.getElementById("lambda_inter").value);
    document.getElementById("lambda_inter_val").textContent = "λ = " + lambda;

    let inter = simulateInterarrivals(lambda);

    let x = [], y = [];
    let maxx = Math.max(...inter, 1);
    for (let t=0; t<=maxx; t+=maxx/200) {
        x.push(t);
        y.push(lambda * Math.exp(-lambda*t));
    }

    Plotly.newPlot("histInter", [
        {
            x: inter,
            type: "histogram",
            opacity: 0.6,
            name: "Interarrival samples"
        },
        {
            x: x,
            y: y,
            mode: "lines",
            name: "λ e^{-λx}"
        }
    ], {
        title: "Interarrival Times vs Exponential(λ)",
        xaxis: {title: "x"},
        yaxis: {title: "Density (scaled)"}
    });
}

    genInter();

## Summary

- Simulating a Bernoulli event in each of $$ ( n ) $$ small intervals with probability $$ ( \lambda/n ) $$ produces a discrete counting process that converges to a Poisson process with rate $$ ( \lambda ) $$. The limiting process has Poisson-distributed increments, independent increments, and exponentially distributed inter-arrival times.

### ***Notes***:
- Independent increments  
- Stationary increments  
- (N(t+h)-N(t) $$\sim \text{Poisson} $$  $$ (\lambda_h) $$)  
- Interarrival times are Exponential ( $$ (\lambda ) $$)

$$
\mathbb{E}[N(1)] = \lambda.
$$

- Higher $$ (\lambda) $$ ⇒ more frequent events.  
- Lower $$ (\lambda) $$ ⇒ fewer events.
